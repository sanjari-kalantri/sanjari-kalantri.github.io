% \placelastupdatedtext prints a grey "Last updated ..." in the top right.
% build.py defines \lastupdatedtext and calls this only when cv.yml asks for it.
\newcommand{\placelastupdatedtext}{%
  \AddToShipoutPictureFG*{%
    \put(
        \LenToUnit{\paperwidth-2 cm-0.2 cm+0.05cm},
        \LenToUnit{\paperheight-1.0 cm}
    ){\vtop{{\null}\makebox[0pt][c]{%
        \small\color{gray}\textit{\lastupdatedtext}\hspace{\widthof{\lastupdatedtext}}
    }}}%
  }%
}%
